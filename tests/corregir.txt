def generate_realistic_test_data():
    """Genera datos con propiedades estadísticas reales."""
    np.random.seed(42)  # Reproducible

    prices = [100]
    for _ in range(1000):
        # Geometric Brownian Motion (más realista que random)
        ret = np.random.normal(0.0005, 0.02)  # Drift + volatility
        prices.append(prices[-1] * (1 + ret))

    quotes = []
    for i, price in enumerate(prices):
        quotes.append(Quote(
            symbol="AAPL",
            timestamp=datetime(2020,1,1) + timedelta(days=i),
            close=Decimal(str(round(price, 2))),
            # Bid/Ask con spread realista
            bid=Decimal(str(round(price * 0.9995, 2))),
            ask=Decimal(str(round(price * 1.0005, 2))),
            volume=Decimal(str(int(np.random.lognormal(17, 0.5)))),  # Volumen lognormal
        ))

    return quotes
