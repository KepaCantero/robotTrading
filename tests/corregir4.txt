  2. Datos Realistas (GBM)
  # ❌ ANTES (Malo)
  for i in range(252):
      price = 100 + i * 0.1  # Tendencia lineal
      volume = 50000000  # Constante

  # ✅ DESPUÉS (Bueno)
  def generate_realistic_quotes(days=252, drift=0.0005, volatility=0.02):
      quotes = []
      price = 100.0

      for i in range(days):
          # Geometric Brownian Motion
          ret = np.random.normal(drift, volatility)
          price = price * (1 + ret)

          # Volumen correlacionado con volatilidad
          vol_mult = 1 + abs(ret) * 10
          volume = 50_000_000 * vol_mult * np.random.lognormal(0, 0.3)

          # Spread realista
          spread_bps = 5 + abs(ret) * 100

          quotes.append(Quote(
              symbol="AAPL",
              close=Decimal(str(round(price, 2))),
              volume=Decimal(str(int(volume))),
              bid=Decimal(str(round(price * (1 - spread_bps/20000), 2))),
              ask=Decimal(str(round(price * (1 + spread_bps/20000), 2))),
          ))

      return quotes
