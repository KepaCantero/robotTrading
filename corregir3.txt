python@pytest.fixture
def realistic_quotes():
    """Generate realistic quotes with market characteristics."""
    quotes = []
    base_price = 100.0
    base_volume = 50_000_000
    start_date = datetime(2020, 1, 1)

    # Simular regímenes de mercado
    trend = 0.0005  # Drift
    volatility = 0.02

    for i in range(252):
        # Geometric Brownian Motion (más realista)
        ret = np.random.normal(trend, volatility)
        price = base_price * (1 + ret)
        base_price = price

        # Volumen correlacionado con volatilidad
        vol_multiplier = 1 + abs(ret) * 10  # Mayor volatilidad = más volumen
        volume = base_volume * vol_multiplier * np.random.lognormal(0, 0.3)

        # Spread realista (bid-ask spread aumenta con volatilidad)
        spread_bps = 5 + abs(ret) * 100  # 5-15 bps típico
        bid = price * (1 - spread_bps / 20000)
        ask = price * (1 + spread_bps / 20000)

        # OHLC coherente
        daily_range = abs(np.random.normal(0, volatility)) * price
        high = price + daily_range / 2
        low = price - daily_range / 2

        quotes.append(Quote(
            symbol="AAPL",
            timestamp=start_date + timedelta(days=i),
            bid=Decimal(str(round(bid, 2))),
            ask=Decimal(str(round(ask, 2))),
            last=Decimal(str(round(price, 2))),
            volume=Decimal(str(int(volume))),
            open=Decimal(str(round(price - ret * price, 2))),
            high=Decimal(str(round(high, 2))),
            low=Decimal(str(round(low, 2))),
            close=Decimal(str(round(price, 2))),
        ))

    return quotes
